// ============================================================================
// UTILITAIRES DE FORMATAGE
// ============================================================================

const formatMoney = (amount, currency) => {
  if (currency === 'GDS') {
    return `${Number(amount).toLocaleString('fr-FR', { maximumFractionDigits: 2 })} GDS`;
  }
  return `${Number(amount).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} $USD`;
};

const calculateEndDate = (startDateStr, monthsToAdd) => {
  const date = new Date(startDateStr);
  date.setMonth(date.getMonth() + monthsToAdd);
  return date.toLocaleDateString('fr-FR', {
    day: '2-digit',
    month: 'short',
    year: 'numeric',
  });
};
